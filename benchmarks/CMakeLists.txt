set(BENCHMARKS throughput push_pop prodcon bfs)

foreach(benchmark ${BENCHMARKS})
  add_library(${benchmark}_base INTERFACE)
  target_sources(${benchmark}_base INTERFACE ${benchmark}.cpp)
  target_link_libraries(${benchmark}_base INTERFACE benchmark_base)
  add_custom_target(${benchmark}_all)
endforeach()
foreach(benchmark ${LOG_BENCHMARKS})
  add_custom_target(${benchmark}_log_all)
endforeach()

function(add_benchmark benchmark target)
  add_executable(${benchmark}_${target})
  target_link_libraries(${benchmark}_${target} PRIVATE ${target} ${benchmark}_base)
  add_dependencies(${benchmark}_all ${benchmark}_${target})
  if(benchmark IN_LIST LOG_BENCHMARKS)
    add_executable(${benchmark}_log_${target})
    target_link_libraries(${benchmark}_log_${target} PRIVATE ${target} ${benchmark}_base)
    target_compile_definitions(${benchmark}_log_${target} PRIVATE LOG_OPERATIONS)
    add_dependencies(${benchmark}_log_all ${benchmark}_log_${target})
  endif()
endfunction()

foreach(target ${COMPETITORS})
  foreach(benchmark ${BENCHMARKS})
    add_benchmark(${benchmark} ${target})
  endforeach()
endforeach()
